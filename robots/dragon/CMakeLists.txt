cmake_minimum_required(VERSION 3.16)
project(dragon)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(gimbalrotor REQUIRED)
find_package(pluginlib REQUIRED)
find_package(aerial_robot_model REQUIRED)
add_library(dragon_robot_model SHARED src/dragon_robot_model.cpp)
ament_target_dependencies(dragon_robot_model gimbalrotor pluginlib aerial_robot_model)
pluginlib_export_plugin_description_file(aerial_robot_model plugins/robot_model_plugins.xml)
install(TARGETS dragon_robot_model LIBRARY DESTINATION lib)
install(DIRECTORY config launch robots urdf plugins DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/transformation_demo.py DESTINATION lib/${PROJECT_NAME})
ament_package()
